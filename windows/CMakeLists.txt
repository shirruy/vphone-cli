# The pinned bzip2 sources expect bz_version.h to be generated by their
# top-level CMake project. We intentionally bypass that project to avoid its
# policy/dependency side effects, so generate the one required header here.
set(VPHONE_BZIP2_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated/bzip2")
file(MAKE_DIRECTORY "${VPHONE_BZIP2_GENERATED_DIR}")
set(BZ_VERSION "1.1.0")
configure_file(
    "${bzip2_SOURCE_DIR}/bz_version.h.in"
    "${VPHONE_BZIP2_GENERATED_DIR}/bz_version.h"
    @ONLY
)

add_library(vphone_bzip2_static STATIC
    "${bzip2_SOURCE_DIR}/blocksort.c"
    "${bzip2_SOURCE_DIR}/huffman.c"
    "${bzip2_SOURCE_DIR}/crctable.c"
    "${bzip2_SOURCE_DIR}/randtable.c"
    "${bzip2_SOURCE_DIR}/compress.c"
    "${bzip2_SOURCE_DIR}/decompress.c"
    "${bzip2_SOURCE_DIR}/bzlib.c"
)
target_include_directories(vphone_bzip2_static PUBLIC
    "${bzip2_SOURCE_DIR}"
    "${VPHONE_BZIP2_GENERATED_DIR}"
)
target_compile_definitions(vphone_bzip2_static PUBLIC BZ2_STATICLIB)
